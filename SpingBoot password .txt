package com.example.formapp.controller;


import com.example.formapp.model.Person;
import org.springframework.stereotype.Controller;
import org.springframework.ui.Model;
import org.springframework.validation.BindingResult;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.ModelAttribute;


@Controller
public class FormController {


    @GetMapping("/form")
    public String showForm(Model model) {
        model.addAttribute("person", new Person());
        return "form";
    }


    @PostMapping("/submitForm")
    public String submitForm(@ModelAttribute Person person, BindingResult bindingResult, Model model) {
        // Verifica se la password e la conferma sono uguali
        if (!person.getPassword().equals(person.getConfirmPassword())) {
            // Se le password non corrispondono, registriamo l'errore
            bindingResult.rejectValue("confirmPassword", "error.person", "Le password non corrispondono");
        }


        // Se ci sono errori, ritorna al form con gli errori
        if (bindingResult.hasErrors()) {
            return "form";
        }


        // Se non ci sono errori, mostra il risultato
        model.addAttribute("person", person);
        return "result";
    }






<!DOCTYPE html>
<html lang="it">
<head>
    <meta charset="UTF-8">
    <meta http-equiv="X-UA-Compatible" content="IE=edge">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Form di Registrazione</title>
</head>
<body>
    <h2>Inserisci i tuoi dati</h2>


    <!-- Mostra gli errori, se ci sono -->
    <div th:if="${#fields.hasErrors()}">
        <ul>
            <li th:each="err : ${#fields.errors()}">
                <span th:text="${err}"></span>
            </li>
        </ul>
    </div>


    <form action="/submitForm" method="post">
        <label for="nome">Nome:</label>
        <input type="text" id="nome" name="nome" th:value="${person.nome}" required><br><br>


        <label for="cognome">Cognome:</label>
        <input type="text" id="cognome" name="cognome" th:value="${person.cognome}" required><br><br>


        <label for="eta">Età:</label>
        <input type="number" id="eta" name="eta" th:value="${person.eta}" required><br><br>


        <label for="annoNascita">Anno di Nascita:</label>
        <input type="number" id="annoNascita" name="annoNascita" th:value="${person.annoNascita}" required><br><br>


        <label for="password">Password:</label>
        <input type="password" id="password" name="password" th:value="${person.password}" required><br><br>


        <label for="confirmPassword">Conferma Password:</label>
        <input type="password" id="confirmPassword" name="confirmPassword" th:value="${person.confirmPassword}" required><br><br>


        <button type="submit">Invia</button>
    </form>
</body>
</html>
